def butter_filter(data, cutoff, fs, order=4, btype='low'):
    nyq = 0.5 * fs
    if isinstance(cutoff, list):
        normal_cutoff = [c / nyq for c in cutoff]
    else:
        normal_cutoff = cutoff / nyq
    b, a = signal.butter(order, normal_cutoff, btype=btype, analog=False)
    return signal.filtfilt(b, a, data)


def get_slope(series):
    if len(series) < 2: return 0
    return stats.linregress(np.arange(len(series)), series)[0]



def estimate_respiration_from_bvp(peaks, ibi, fs):

    if len(peaks) < 4: return 0

    t_ibi = np.cumsum(ibi) / 1000.0
    if t_ibi[-1] <= t_ibi[0]: return 0

    f_interp = interpolate.interp1d(t_ibi, ibi, kind='cubic', fill_value="extrapolate")
    fs_resp = 4.0
    duration = t_ibi[-1] - t_ibi[0]
    if duration < 2: return 0

    steps = np.arange(0, duration, 1 / fs_resp)
    ibi_interp = f_interp(steps)


    f, psd = signal.welch(ibi_interp, fs=fs_resp, nperseg=len(ibi_interp) // 2)

    resp_band_mask = (f >= 0.1) & (f <= 0.5)
    if np.sum(resp_band_mask) == 0: return 0

    f_band = f[resp_band_mask]
    psd_band = psd[resp_band_mask]

    dominant_freq = f_band[np.argmax(psd_band)]
    return dominant_freq


def get_eda_peak_features(phasic_data, fs):

    threshold = np.std(phasic_data) * 0.5
    peaks, _ = signal.find_peaks(phasic_data, height=threshold, distance=fs * 2)

    num_peaks = len(peaks)
    mean_peak_height = np.mean(phasic_data[peaks]) if num_peaks > 0 else 0

    return [num_peaks, mean_peak_height]


def get_nonlinear_hrv_features(ibi_ms):
    if len(ibi_ms) < 2: return [0, 0, 0]
    diff_ibi = np.diff(ibi_ms)
    sd1 = np.sqrt(np.std(diff_ibi, ddof=1) ** 2 * 0.5)
    sd2 = np.sqrt(2 * np.std(ibi_ms, ddof=1) ** 2 - 0.5 * np.std(diff_ibi, ddof=1) ** 2)
    ratio = sd1 / sd2 if sd2 > 0 else 0
    return [sd1, sd2, ratio]


def get_frequency_domain_features(ibi_ms):
    if len(ibi_ms) < 2: return [0, 0, 0, 0]
    t_ibi = np.cumsum(ibi_ms) / 1000.0
    t_ibi = t_ibi - t_ibi[0]
    if t_ibi[-1] < 1.0: return [0, 0, 0, 0]
    f_interp = interpolate.interp1d(t_ibi, ibi_ms, kind='cubic', fill_value="extrapolate")
    fs_interp = 4.0
    steps = np.arange(0, t_ibi[-1], 1 / fs_interp)
    if len(steps) < 10: return [0, 0, 0, 0]
    ibi_interp = f_interp(steps)
    f, psd = signal.welch(ibi_interp, fs=fs_interp, nperseg=len(ibi_interp) // 2)

    vlf_power = np.trapz(psd[(f >= 0.003) & (f < 0.04)], f[(f >= 0.003) & (f < 0.04)])
    lf_power = np.trapz(psd[(f >= 0.04) & (f < 0.15)], f[(f >= 0.04) & (f < 0.15)])
    hf_power = np.trapz(psd[(f >= 0.15) & (f < 0.4)], f[(f >= 0.15) & (f < 0.4)])
    ratio = lf_power / hf_power if hf_power > 1e-6 else 0
    return [vlf_power, lf_power, hf_power, ratio]


def extract_features_physiological(window_data, fs, sensor_type):
    if len(window_data) == 0: return [0] * 5

    mean_val = np.mean(window_data)
    std_val = np.std(window_data)
    iqr_val = iqr(window_data)
    feats = [mean_val, std_val, iqr_val]

    if sensor_type == 'EDA':
        slope = get_slope(window_data)
        feats.append(slope)

        phasic = butter_filter(window_data, 0.05, fs, btype='high', order=2)
        phasic_std = np.std(phasic)
        line_length = np.sum(np.abs(np.diff(phasic)))

        peak_feats = get_eda_peak_features(phasic, fs)

        feats.extend([phasic_std, line_length])
        feats.extend(peak_feats)  # [num_peaks, mean_peak_height]
        feats.append(np.max(window_data) - np.min(window_data))

    elif sensor_type == 'BVP':
        clean_bvp = butter_filter(window_data, [0.5, 8.0], fs, btype='band')
        peaks, _ = signal.find_peaks(clean_bvp, distance=fs / 2.5)

        if len(peaks) > 1:
            ibi = np.diff(peaks) / fs * 1000
            mean_hr = 60000 / np.mean(ibi)
            rmssd = np.sqrt(np.mean(np.square(np.diff(ibi))))
            sdnn = np.std(ibi)
            pnn50 = np.sum(np.abs(np.diff(ibi)) > 50) / len(ibi)

            freq_feats = get_frequency_domain_features(ibi)
            nonlinear_feats = get_nonlinear_hrv_features(ibi)

            resp_rate = estimate_respiration_from_bvp(peaks, ibi, fs)

            feats.extend([mean_hr, rmssd, sdnn, pnn50])
            feats.extend(freq_feats)
            feats.extend(nonlinear_feats)
            feats.append(resp_rate)
        else:
            feats.extend([0] * 12)

    elif sensor_type == 'TEMP':
        slope = get_slope(window_data)
        feats.append(slope)

    return feats



def process_subject(subject_path, subject_id):
    if not os.path.exists(subject_path): return None
    with open(subject_path, 'rb') as file:
        data = pickle.load(file, encoding='latin1')

    wrist_data = data['signal']['wrist']
    labels = data['label']
    duration_sec = len(labels) / FS_DICT['LABEL']
    starts = np.arange(0, duration_sec - WINDOW_SIZE, STEP_SIZE)

    X_raw, y_raw = [], []

    for start in starts:
        end = start + WINDOW_SIZE
        idx_s_lbl = int(start * FS_DICT['LABEL'])
        idx_e_lbl = int(end * FS_DICT['LABEL'])
        w_lbls = labels[idx_s_lbl:idx_e_lbl]
        if len(w_lbls) == 0: continue
        mode_label = stats.mode(w_lbls, keepdims=True)[0][0]
        if mode_label not in TARGET_LABELS: continue

        w_feats = []
        # EDA
        s, e = int(start * 4), int(end * 4)
        w_feats.extend(extract_features_physiological(wrist_data['EDA'][s:e].flatten(), 4, 'EDA'))
        # BVP
        s, e = int(start * 64), int(end * 64)
        w_feats.extend(extract_features_physiological(wrist_data['BVP'][s:e].flatten(), 64, 'BVP'))
        # TEMP
        s, e = int(start * 4), int(end * 4)
        w_feats.extend(extract_features_physiological(wrist_data['TEMP'][s:e].flatten(), 4, 'TEMP'))
        # ACC
        s, e = int(start * 32), int(end * 32)
        acc_mag = np.linalg.norm(wrist_data['ACC'][s:e], axis=1)
        w_feats.append(np.std(acc_mag))

        X_raw.append(w_feats)
        y_raw.append(mode_label)

    if len(X_raw) == 0: return None
    df_subj = pd.DataFrame(X_raw)
    df_subj['label_raw'] = y_raw

    # Subject-Specific Normalization
    baseline_data = df_subj[df_subj['label_raw'] == 1]
    feat_cols_to_norm = df_subj.columns[:-1]
    if len(baseline_data) > 0:
        base_mean = baseline_data[feat_cols_to_norm].mean()
        df_subj[feat_cols_to_norm] = df_subj[feat_cols_to_norm] - base_mean
    else:
        df_subj[feat_cols_to_norm] = df_subj[feat_cols_to_norm] - df_subj[feat_cols_to_norm].mean()

    df_subj['label'] = df_subj['label_raw'].map(LABEL_MAP)
    df_subj['subject'] = subject_id
    df_subj.drop(columns=['label_raw'], inplace=True)
    return df_subj


def load_dataset():
    subjects = ['S2', 'S3', 'S4', 'S5', 'S6', 'S7', 'S8', 'S9', 'S10', 'S11', 'S13', 'S14', 'S15', 'S16', 'S17']
    dfs = []
    print("开始加载数据...")
    for s in subjects:
        p = os.path.join(WESAD_ROOT_PATH, s, f'{s}.pkl')
        print(f"Processing {s}...", end='\r')
        res = process_subject(p, s)
        if res is not None: dfs.append(res)
    print("\n数据加载完成。")
    return pd.concat(dfs, ignore_index=True)


if __name__ == "__main__":
    full_df = load_dataset()

    test_subject = 'S17'
    train_df = full_df[full_df['subject'] != test_subject]
    test_df = full_df[full_df['subject'] == test_subject]

    train_df = train_df.fillna(0)
    test_df = test_df.fillna(0)

    feature_cols = [c for c in train_df.columns if c not in ['label', 'subject']]
    X_train, y_train = train_df[feature_cols], train_df['label']
    X_test, y_test = test_df[feature_cols], test_df['label']

    print(f"Features Count: {len(feature_cols)}")

    pipeline = Pipeline([
        ('scaler', RobustScaler()),
        ('clf', GradientBoostingClassifier(
            n_estimators=600,
            learning_rate=0.04,
            max_depth=4,
            min_samples_leaf=15,
            subsample=0.85,
            random_state=42
        ))
    ])

    print("开始训练...")
    pipeline.fit(X_train, y_train)

    print("正在预测...")
    y_pred = pipeline.predict(X_test)
    y_pred_smooth = signal.medfilt(y_pred, kernel_size=5)

    print(f"\n=== {test_subject} 最终测试结果 ===")
    print(classification_report(y_test, y_pred_smooth, target_names=TARGET_NAMES))
    print("混淆矩阵:")
    print(confusion_matrix(y_test, y_pred_smooth))

        if st.session_state.continuous_state in [1, 3] and st.session_state.continuous_count >= 15:
            state_name = "Overwhelmed" if st.session_state.continuous_state == 1 else "Zoned Out"
            elapsed_seconds = st.session_state.continuous_count * 2
            banner_html = f"""
            

                

                    ⚠️
                    ALERT: The child's {state_name} state has lasted for 
                    
                        {elapsed_seconds}
                     seconds! Please take action.
                

            

            """
            st.markdown(banner_html, unsafe_allow_html=True)

        st.session_state.history_time.append(current_time_dt)
        st.session_state.history_pred_display.append(st.session_state.current_display_state)
        st.session_state.history_hr.append(current_hr)
        st.session_state.history_eda.append(current_eda)
        st.session_state.history_temp.append(current_temp)

        card_html, btn_text = get_kori_ui_elements(st.session_state.current_display_state)
        st.markdown(card_html, unsafe_allow_html=True)

        # Button click logic: Trigger the hardware device to start working
        if st.button(f"🗜️ {btn_text}", key="hardware_action_btn", use_container_width=True):
            if hardware_serial is not None and hardware_serial.is_open:
                hardware_serial.write(b'1')
                print("Hardware device triggered and is now working.")
            else:
                st.error("Hardware device is not connected. Cannot start operation!")

        fig = make_subplots(
            rows=4, cols=1, shared_xaxes=True, vertical_spacing=0.08,
            row_heights=[0.1, 0.3, 0.3, 0.3],
            subplot_titles=("", "Heart Rate", "EDA", "Skin Temp")
        )

        colors_for_timeline = [COLOR_MAP[p] for p in st.session_state.history_pred_display]
        fig.add_trace(go.Scatter(
            x=list(st.session_state.history_time), y=[1] * len(st.session_state.history_time),
            mode='markers+lines', line=dict(color='#EFEFEF', width=10),
            marker=dict(size=20, color=colors_for_timeline, symbol='square'),
            showlegend=False
        ), row=1, col=1)

        fig.add_trace(go.Scatter(
            x=list(st.session_state.history_time), y=list(st.session_state.history_hr),
            mode='lines', line=dict(color='#C692E9', width=2, shape='spline'), showlegend=False
        ), row=2, col=1)

        fig.add_trace(go.Scatter(
            x=list(st.session_state.history_time), y=list(st.session_state.history_eda),
            mode='lines', line=dict(color='#43C4C9', width=2, shape='spline'), showlegend=False
        ), row=3, col=1)

        fig.add_trace(go.Scatter(
            x=list(st.session_state.history_time), y=list(st.session_state.history_temp),
            mode='lines', line=dict(color='#F28C50', width=2, shape='spline'), showlegend=False
        ), row=4, col=1)

        fig.update_layout(height=600, margin=dict(l=10, r=10, t=30, b=10), plot_bgcolor='white', paper_bgcolor='white')
        fig.update_yaxes(showticklabels=False, showgrid=False, zeroline=False, row=1, col=1)
        for r in range(2, 5):
            fig.update_yaxes(showgrid=True, gridcolor='#F0F0F0', griddash='dash', zeroline=False, row=r, col=1)
        fig.update_xaxes(showgrid=True, gridcolor='#F0F0F0', griddash='dash', zeroline=False, tickformat="%H:%M:%S")

        st.plotly_chart(fig, use_container_width=True)

        st.session_state.step += 1
        time.sleep(1)

        st.rerun()
    else:
        st.success("Simulation stream ended!")
        st.session_state.is_running = False
